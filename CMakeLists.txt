cmake_minimum_required(VERSION 3.20)
project(TaskForge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PostgreSQL REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.11 REQUIRED)
add_library(taskforge_core INTERFACE)
target_include_directories(taskforge_core INTERFACE services/core/include)
target_link_libraries(taskforge_core INTERFACE PostgreSQL::PostgreSQL Threads::Threads nlohmann_json::nlohmann_json)
target_compile_options(taskforge_core INTERFACE -Wall -Wextra -Wpedantic -Werror)
set(TASKFORGE_SANITIZER "" CACHE STRING "Optional address or thread sanitizer")
if(TASKFORGE_SANITIZER)
 if(NOT TASKFORGE_SANITIZER MATCHES "^(address|thread)$")
  message(FATAL_ERROR "Use address or thread sanitizer")
 endif()
 target_compile_options(taskforge_core INTERFACE -fsanitize=${TASKFORGE_SANITIZER} -fno-omit-frame-pointer)
 target_link_options(taskforge_core INTERFACE -fsanitize=${TASKFORGE_SANITIZER})
endif()
add_executable(taskforge-scheduler services/scheduler/main.cpp)
target_link_libraries(taskforge-scheduler PRIVATE taskforge_core)
add_executable(taskforge-worker services/worker/main.cpp)
target_link_libraries(taskforge-worker PRIVATE taskforge_core)
